7impl SessionBuilder {
8	/// Enable/disable the usage of prepacking.
9	///
10	/// This option is **enabled** by default.
11	pub fn with_prepacking(self, enable: bool) -> BuilderResult {
12		self.with_config_entry("session.disable_prepacking", if enable { "0" } else { "1" })
13	}
14
15	/// Use allocators from the registered environment.
16	///
17	/// This option is **disabled** by default.
18	pub fn with_env_allocators(self) -> BuilderResult {
19		self.with_config_entry("session.use_env_allocators", "1")
20	}
21
22	/// Disables subnormal floats by enabling the denormals-are-zero and flush-to-zero flags for all threads in the
23	/// session's internal thread pool.
24	///
25	/// [Subnormal floats](https://en.wikipedia.org/wiki/Subnormal_number) are extremely small numbers very close to zero.
26	/// Operations involving subnormal numbers can be very slow; enabling this flag will instead treat them as `0.0`,
27	/// giving faster & more consistent performance, but lower accuracy (in cases where subnormals are involved).
28	///
29	/// This option is **disabled** by default, as it may hurt model accuracy.
30	pub fn with_flush_to_zero(self) -> BuilderResult {
31		self.with_config_entry("session.set_denormal_as_zero", "1")
32	}
33
34	/// Enable/disable fusion for quantized models in QDQ (`QuantizeLinear`/`DequantizeLinear`) format.
35	///
36	/// This option is **enabled** by default for all EPs except DirectML.
37	pub fn with_quant_qdq(self, enable: bool) -> BuilderResult {
38		self.with_config_entry("session.disable_quant_qdq", if enable { "0" } else { "1" })
39	}
40
41	/// Enable/disable the optimization step removing double QDQ nodes.
42	///
43	/// This option is **enabled** by default.
44	pub fn with_double_qdq_remover(self, enable: bool) -> BuilderResult {
45		self.with_config_entry("session.disable_double_qdq_remover", if enable { "0" } else { "1" })
46	}
47
48	/// Enable the removal of Q/DQ node pairs once all QDQ handling has been completed.
49	///
50	/// This option is **disabled** by default.
51	pub fn with_qdq_cleanup(self) -> BuilderResult {
52		self.with_config_entry("session.enable_quant_qdq_cleanup", "1")
53	}
54
55	/// Enable fast tanh-based GELU approximation (like PyTorch's `nn.GELU(approximate='tanh')`).
56	///
57	/// This option is **disabled** by default, as it may impact results.
58	pub fn with_approximate_gelu(self) -> BuilderResult {
59		self.with_config_entry("optimization.enable_gelu_approximation", "1")
60	}
61
62	/// Enable the `Cast` chain elimination optimization.
63	///
64	/// This option is **disabled** by default, as it may impact results.
65	pub fn with_cast_chain_elimination(self) -> BuilderResult {
66		self.with_config_entry("optimization.enable_cast_chain_elimination", "1")
67	}
68
69	/// Enable/disable ahead-of-time function inlining.
70	///
71	/// This option is **enabled** by default.
72	pub fn with_aot_inlining(self, enable: bool) -> BuilderResult {
73		self.with_config_entry("session.disable_aot_function_inlining", if enable { "0" } else { "1" })
74	}
75
76	/// Accepts a comma-separated list of optimizers to disable.
77	pub fn with_disabled_optimizers(self, optimizers: impl AsRef<str>) -> BuilderResult {
78		self.with_config_entry("optimization.disable_specified_optimizers", optimizers)
79	}
80
81	/// Enable using the device allocator for allocating initialized tensor memory, potentially bypassing arena
82	/// allocators.
83	///
84	/// This option is **disabled** by default.
85	pub fn with_device_allocated_initializers(self) -> BuilderResult {
86		self.with_config_entry("session.use_device_allocator_for_initializers", "1")
87	}
88
89	/// Enable/disable allowing the inter-op threads to spin for a short period before blocking.
90	///
91	/// This option is **enabled** by defualt.
92	pub fn with_inter_op_spinning(self, enable: bool) -> BuilderResult {
93		self.with_config_entry("session.inter_op.allow_spinning", if enable { "1" } else { "0" })
94	}
95
96	/// Enable/disable allowing the intra-op threads to spin for a short period before blocking.
97	///
98	/// This option is **enabled** by defualt.
99	pub fn with_intra_op_spinning(self, enable: bool) -> BuilderResult {
100		self.with_config_entry("session.intra_op.allow_spinning", if enable { "1" } else { "0" })
101	}
102
103	/// Disables falling back to the CPU for operations not supported by any other EP.
104	/// Models with graphs that cannot be placed entirely on the EP(s) will fail to commit.
105	pub fn with_disable_cpu_fallback(self) -> BuilderResult {
106		self.with_config_entry("session.disable_cpu_ep_fallback", "1")
107	}
108
109	/// Uses slower U8U8 matrix multiplication in place of U8S8 matrix multiplication that could potentially overflow on
110	/// x86-64 platforms without the VNNI extension.
111	///
112	/// This should only be enabled if you encounter overflow issues with quantized models.
113	pub fn with_precise_qmm(self) -> BuilderResult {
114		self.with_config_entry("session.x64quantprecision", "1")
115	}
116
117	/// Enables dynamic thread block sizing with the given base block size.
118	pub fn with_dynamic_block_base(self, size: u32) -> BuilderResult {
119		self.with_config_entry("session.dynamic_block_base", size.to_string())
120	}
121}
