1use core::{cell::UnsafeCell, convert::Infallible, marker::PhantomData, mem::MaybeUninit, ptr};
2use std::sync::Once;
3
4pub(crate) struct OnceLock<T> {
5	data: UnsafeCell<MaybeUninit<T>>,
6	once: std::sync::Once,
7	phantom: PhantomData<T>
8}
9
10unsafe impl<T: Send> Send for OnceLock<T> {}
11unsafe impl<T: Send + Sync> Sync for OnceLock<T> {}
12
13impl<T> OnceLock<T> {
14	pub const fn new() -> Self {
15		Self {
16			data: UnsafeCell::new(MaybeUninit::uninit()),
17			once: Once::new(),
18			phantom: PhantomData
19		}
20	}
21
22	#[inline]
23	pub fn get_or_init<F: FnOnce() -> T>(&self, f: F) -> &T {
24		match self.get_or_try_init(|| Ok::<T, Infallible>(f())) {
25			Ok(x) => x,
26			Err(e) => match e {}
27		}
28	}
29
30	#[inline]
31	pub fn get(&self) -> Option<&T> {
32		if self.once.is_completed() { Some(unsafe { self.get_unchecked() }) } else { None }
33	}
34
35	#[inline]
36	pub unsafe fn get_unchecked(&self) -> &T {
37		unsafe { &*(*self.data.get()).as_ptr() }
38	}
39
40	#[inline]
41	pub fn get_or_try_init<F: FnOnce() -> Result<T, E>, E>(&self, f: F) -> Result<&T, E> {
42		if let Some(value) = self.get() { Ok(value) } else { self.try_init_inner(f) }
43	}
44
45	#[cold]
46	fn try_init_inner<F: FnOnce() -> Result<T, E>, E>(&self, f: F) -> Result<&T, E> {
47		let mut res: Result<(), E> = Ok(());
48		let slot = &self.data;
49		self.once.call_once_force(|_| match f() {
50			Ok(value) => unsafe {
51				(*slot.get()).write(value);
52			},
53			Err(e) => {
54				res = Err(e);
55			}
56		});
57		res.map(|_| unsafe { self.get_unchecked() })
58	}
59}
60
61impl<T> OnceLock<T> {
62	pub fn try_insert_with<F: FnOnce() -> T>(&self, inserter: F) -> bool {
63		let mut container = Some(inserter);
64		self.get_or_init(|| (unsafe { container.take().unwrap_unchecked() })());
65		container.is_none()
66	}
67}
68
69impl<T> Drop for OnceLock<T> {
70	fn drop(&mut self) {
71		if self.once.is_completed() {
72			unsafe {
73				ptr::drop_in_place((*self.data.get()).as_mut_ptr());
74			}
75		}
76	}
77}
